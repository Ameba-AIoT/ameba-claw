set(private_sources)
set(private_includes)

set(BOARD "" CACHE STRING
    "Default board for cap_board_mgr. Empty = first alphabetically. \
Pass via: python ameba.py build -a <app> -D BOARD=<name>")

# ---- Scan chip and board JSON files ----
set(CHIP_JSON_DIR  ${CMAKE_CURRENT_SOURCE_DIR}/chips)
set(BOARD_JSON_DIR ${CMAKE_CURRENT_SOURCE_DIR}/boards)

file(GLOB        CHIP_JSON_FILES       ${CHIP_JSON_DIR}/*.json)
file(GLOB_RECURSE ALL_BOARD_JSON_FILES ${BOARD_JSON_DIR}/*/board.json)

if(NOT ALL_BOARD_JSON_FILES)
    message(FATAL_ERROR "[cap_board_mgr] No board.json found under ${BOARD_JSON_DIR}")
endif()

list(SORT ALL_BOARD_JSON_FILES)

# Build parallel list of board names (same order as ALL_BOARD_JSON_FILES)
set(AVAILABLE_BOARD_NAMES "")
foreach(BOARD_FILE ${ALL_BOARD_JSON_FILES})
    get_filename_component(BOARD_DIR  ${BOARD_FILE} DIRECTORY)
    get_filename_component(BOARD_NAME ${BOARD_DIR}  NAME)
    list(APPEND AVAILABLE_BOARD_NAMES "${BOARD_NAME}")
endforeach()

# ---- Select default board ----
if(BOARD)
    list(FIND AVAILABLE_BOARD_NAMES "${BOARD}" BOARD_IDX)
    if(BOARD_IDX EQUAL -1)
        message(FATAL_ERROR
            "[cap_board_mgr] BOARD='${BOARD}' not found.\n"
            "  Available boards: ${AVAILABLE_BOARD_NAMES}\n"
            "  Usage: python ameba.py build -a <app> -D BOARD=<name>")
    endif()
    list(GET ALL_BOARD_JSON_FILES ${BOARD_IDX} DEFAULT_BOARD_JSON_FILE)
    set(DEFAULT_BOARD_NAME "${BOARD}")
else()
    list(GET ALL_BOARD_JSON_FILES    0 DEFAULT_BOARD_JSON_FILE)
    list(GET AVAILABLE_BOARD_NAMES  0 DEFAULT_BOARD_NAME)
endif()

message(STATUS "[cap_board_mgr] Default board: ${DEFAULT_BOARD_NAME}  (available: ${AVAILABLE_BOARD_NAMES})")

# ---- Generated output directory ----
set(BOARD_JSON_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${BOARD_JSON_GEN_DIR})

# ---- Embed all chip JSON files ----
set(BOARD_JSON_ASM "${BOARD_JSON_GEN_DIR}/board_json.S")
file(WRITE "${BOARD_JSON_ASM}" "    .section .rodata\n")

foreach(CHIP_FILE ${CHIP_JSON_FILES})
    get_filename_component(CHIP_NAME ${CHIP_FILE} NAME_WE)
    file(APPEND "${BOARD_JSON_ASM}"
        "    .balign 4\n"
        "    .global _res_chip_${CHIP_NAME}_start\n"
        "_res_chip_${CHIP_NAME}_start:\n"
        "    .incbin \"${CHIP_FILE}\"\n"
        "    .global _res_chip_${CHIP_NAME}_end\n"
        "_res_chip_${CHIP_NAME}_end:\n"
    )
endforeach()

# ---- Embed all board JSON files ----
foreach(BOARD_FILE ${ALL_BOARD_JSON_FILES})
    get_filename_component(BOARD_DIR  ${BOARD_FILE} DIRECTORY)
    get_filename_component(BOARD_NAME ${BOARD_DIR}  NAME)
    file(APPEND "${BOARD_JSON_ASM}"
        "    .balign 4\n"
        "    .global _res_board_${BOARD_NAME}_start\n"
        "_res_board_${BOARD_NAME}_start:\n"
        "    .incbin \"${BOARD_FILE}\"\n"
        "    .global _res_board_${BOARD_NAME}_end\n"
        "_res_board_${BOARD_NAME}_end:\n"
    )
endforeach()

set_source_files_properties("${BOARD_JSON_ASM}" PROPERTIES
    OBJECT_DEPENDS "${ALL_BOARD_JSON_FILES};${CHIP_JSON_FILES}")

ameba_list_append(private_sources "${BOARD_JSON_ASM}")

# ---- Generate board_registry.h ----
# Provides: extern declarations, BOARD_REGISTRY_ENTRIES macro, CLAW_DEFAULT_BOARD_NAME
set(BOARD_REGISTRY_H "${BOARD_JSON_GEN_DIR}/board_registry.h")

file(WRITE  "${BOARD_REGISTRY_H}" "/* AUTO-GENERATED by CMake — do not edit */\n")
file(APPEND "${BOARD_REGISTRY_H}" "#pragma once\n\n")
file(APPEND "${BOARD_REGISTRY_H}" "/* Default board: ${DEFAULT_BOARD_NAME} */\n")
file(APPEND "${BOARD_REGISTRY_H}" "#define CLAW_DEFAULT_BOARD_NAME \"${DEFAULT_BOARD_NAME}\"\n\n")

foreach(BOARD_FILE ${ALL_BOARD_JSON_FILES})
    get_filename_component(BOARD_DIR  ${BOARD_FILE} DIRECTORY)
    get_filename_component(BOARD_NAME ${BOARD_DIR}  NAME)
    file(APPEND "${BOARD_REGISTRY_H}"
        "extern const unsigned char _res_board_${BOARD_NAME}_start[];\n"
        "extern const unsigned char _res_board_${BOARD_NAME}_end[];\n"
    )
endforeach()

file(APPEND "${BOARD_REGISTRY_H}" "\n/* Board registry table — use inside a static initializer */\n")
file(APPEND "${BOARD_REGISTRY_H}" "#define BOARD_REGISTRY_ENTRIES \\\n")
foreach(BOARD_FILE ${ALL_BOARD_JSON_FILES})
    get_filename_component(BOARD_DIR  ${BOARD_FILE} DIRECTORY)
    get_filename_component(BOARD_NAME ${BOARD_DIR}  NAME)
    file(APPEND "${BOARD_REGISTRY_H}"
        "    { \"${BOARD_NAME}\", _res_board_${BOARD_NAME}_start, _res_board_${BOARD_NAME}_end }, \\\n"
    )
endforeach()
file(APPEND "${BOARD_REGISTRY_H}" "    { NULL, NULL, NULL }\n")

ameba_list_append(private_includes "${BOARD_JSON_GEN_DIR}")

# ---- Library sources and includes ----
ameba_list_append(private_sources
    src/cap_board_mgr.c
)
ameba_list_append(private_includes
    include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../claw_modules/claw_cap/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../claw_modules/claw_agent/include
)

ameba_add_internal_library(cap_board_mgr
    p_SOURCES ${private_sources}
    p_INCLUDES ${private_includes}
    p_COMPILE_OPTIONS -Wno-unused-parameter
)
